Google Sheets Integration Setup

Connect this portal directly to your Google Sheet using Google Apps Script so all student submissions automatically append to your spreadsheet in real time with exact timestamps.

Local storage backup is active. When configured, submissions sync directly to Google Sheets.

3-Minute Setup Instructions:

  1. Create a new Google Sheet named "Class 10 Farewell 2026".
  2. In the top menu, click Extensions > Apps Script.
  3. Delete all code in the script editor, paste the script below, and click Save.
  4. Click Deploy > New deployment > Select type Web app.
  5. Set Execute as: Me and Who has access: Anyone.
  6. Click Deploy, copy the Web App URL, and paste it into the field above!
Google Apps Script (Code.gs): 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // Add header row automatically if empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp",
        "Student Name",
        "Section",
        "Phone / WhatsApp",
        "Attending Decision",
        "Preferred Month",
        "Farewell Wishes / Requests",
        "Pass Token"
      ]);
      sheet.getRange(1, 1, 1, 8).setFontWeight("bold").setBackground("#D4AF37");
    }
    
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      data.timestamp || new Date().toLocaleString(),
      data.name,
      data.section || "N/A",
      data.phone || "N/A",
      data.attending,
      data.month,
      data.suggestions || "None",
      data.token || ""
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      message: "RSVP recorded successfully"
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
